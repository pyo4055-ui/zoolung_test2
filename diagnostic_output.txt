function bookingMealText(b){
  if(b.mealType==="none")return "식사하지 않음";
  const time=b.mealStart&&b.mealEnd?` ${b.mealStart}~${b.mealEnd}`:"";
  if(b.mealType==="lunchbox")return `도시락 지참${time}`;
  if(b.mealType==="cafe"){
    const items=(b.cafe?.items||[]).map(x=>`${x.name} × ${x.qty}`).join(", ");
    return `내부 카페 주문${time}${items?` / ${items}`:""}`;
  }
  return "-";
}

function monthOptions(){
  const s=settings();
  let start=new Date(s.bookingStart+"T00:00:00"), end=new Date(s.bookingEnd+"T00:00:00");
  if(isNaN(start))start=new Date();
  if(isNaN(end) || end<start)end=new Date(start.getFullYear()+1,start.getMonth(),1);
  const first=new Date(start.getFullYear(),start.getMonth(),1);
  const last=new Date(end.getFullYear(),end.getMonth(),1);
  const out=[];
  for(let d=new Date(first);d<=last;d=new Date(d.getFullYear(),d.getMonth()+1,1)){
    out.push({y:d.getFullYear(),m:d.getMonth()+1,key:`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}`,label:`${d.getFullYear()}년 ${d.getMonth()+1}월`})
  }
  return out
}

}
function renderVisitDays(){
  const ym=$("visitMonth").value;
  $("visitDay").innerHTML='<option value="">선택</option>';
  if(!ym)return;
  let [y,m]=ym.split("-").map(Number);
  const last=new Date(y,m,0).getDate(), today=new Date(); today.setHours(0,0,0,0);
  const s=settings(), openStart=new Date(s.bookingStart+"T00:00:00"), openEnd=new Date(s.bookingEnd+"T23:59:59");
  for(let d=1;d<=last;d++){

  }
}

  if(b.mealType==="none")return "식사하지 않음";

    return `내부 카페 주문${time}${items?` / ${items}`:""}`;
  }
  return "-";
}

function bookingMealText(b){
  if(b.mealType==="none")return "식사하지 않음";
  const time=b.mealStart&&b.mealEnd?` ${b.mealStart}~${b.mealEnd}`:"";
  if(b.mealType==="lunchbox")return `도시락 지참${time}`;
  if(b.mealType==="cafe"){
    const items=(b.cafe?.items||[]).map(x=>`${x.name} × ${x.qty}`).join(", ");
    return `내부 카페 주문${time}${items?` / ${items}`:""}`;
  }
  return "-";
}
